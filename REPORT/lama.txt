	PROCEDURE AutomaticInsertTable
	IS
		CURSOR getBA
		IS
			SELECT fccode FROM businessunit;
	BEGIN
		FOR data__ IN getBA
		LOOP
			InsertIntoTable (data__.fccode,
							 SYSDATE - 1,
							 SYSDATE - 1,
							 TO_NUMBER (TO_CHAR (SYSDATE, 'YYYY')),
							 1,
							 1);
		END LOOP;
	END AutomaticInsertTable;


	PROCEDURE InsertIntoTable (fcba_			 IN VARCHAR2,
							   date_from_		 IN DATE,
							   date_until_		 IN DATE,
							   budget_year_ 	 IN NUMBER,
							   budget_version_	 IN NUMBER,
							   module_version_	 IN NUMBER)
	IS
	BEGIN


		Table_Normalization (fcba_, date_from_, date_until_);

		COMMIT;

		-- First Step, to Calculate Landclearing Value
		LandClearing_Process (fcba_,
							  date_from_,
							  date_until_,
							  budget_year_,
							  budget_version_,
							  module_version_,
							  'STEP_01');


		COMMIT;

		-- Second Step, to Calculate Production Value
		Production_Process (fcba_,
							date_from_,
							date_until_,
							budget_year_,
							budget_version_,
							module_version_,
							'STEP_02');



		COMMIT;

		-- Third Step, to Calculate Output Value
		Output_Process (fcba_,
						date_from_,
						date_until_,
						budget_year_,
						budget_version_,
						module_version_,
						'STEP_03');
	


		COMMIT;

		-- Fourth Step, to Calculate Handling Value
		Handling_Process (fcba_,
						  date_from_,
						  date_until_,
						  budget_year_,
						  budget_version_,
						  module_version_,
						  'STEP_04');


		COMMIT;


		-- Fifth Step, to Calculate Handling Amount Value
		HandlingAmount_Process (fcba_,
								date_from_,
								date_until_,
								budget_year_,
								budget_version_,
								module_version_,
								'STEP_05');



		COMMIT;

		-- Sixth Step, to Calculate Fertilizer Value
		Fertilizer_Process (fcba_,
							date_from_,
							date_until_,
							budget_year_,
							budget_version_,
							module_version_,
							'STEP_06');



		COMMIT;


		-- Sixth Step, to Calculate Fertilizer Value
		HumanResources_Process (fcba_,
								date_from_,
								date_until_,
								budget_year_,
								budget_version_,
								module_version_,
								'STEP_07');



		COMMIT;


		-- Eight Step, to Calculate Vehicle Group Value
		VehicleGroup_Process (fcba_,
							  date_from_,
							  date_until_,
							  budget_year_,
							  budget_version_,
							  module_version_,
							  'STEP_08');


		COMMIT;

		-- Ninth Step, to Calculate Ending Balance Value
		EndingBalance_Process (fcba_,
							   date_from_,
							   date_until_,
							   budget_year_,
							   budget_version_,
							   module_version_,
							   'STEP_09');



		COMMIT;


	END InsertIntoTable;


	PROCEDURE Table_Normalization (fcba_ IN VARCHAR2, date_from_ IN DATE, date_until_ IN DATE)
	IS
	BEGIN


		DELETE Production_Summary_Tab
		 WHERE ondate BETWEEN date_from_ AND date_until_ AND fcba = fcba_;



	END Table_Normalization;

	PROCEDURE EndingBalance_Process (fcba_			   IN VARCHAR2,
									 date_from_ 	   IN DATE,
									 date_until_	   IN DATE,
									 budget_year_	   IN NUMBER,
									 budget_version_   IN NUMBER,
									 module_version_   IN NUMBER,
									 nStep_ 		   IN VARCHAR2)
	IS
		CURSOR getMaxLineNo
		IS
			SELECT MAX (NVL (lineno, 0)) maxLineNo
			  FROM production_summary_tab
			 WHERE fcba = fcba_;


		CURSOR setData (todate__ IN DATE)
		IS
			SELECT codedesc,
				   fcba,
				   ROWNUM thenumber,
				   stockGenCode JobGenCode,
				   description,
				   unitofmeasurement,
				   orderdata seqnumber,
				   hi,
				   shi,
				   sbi,
				   0 budget_hi,
				   0 budget_shi,
				   0 budget_sbi
			  FROM (  SELECT fcba,
							 codedesc,
							 stockGenCode,
							 description,
							 orderdata,
							 UnitOfMeasurement,
							 ROUND (SUM (NVL (hi, 0)), 2) hi,
							 ROUND (SUM (NVL (shi, 0)), 2) shi,
							 ROUND (SUM (NVL (sbi, 0)), 2) sbi
						FROM (	SELECT fcba,
									   '1' nomer,
									   'SR003_09' codedesc,
									   stockGenCode,
									   stockGenName description,
									   orderdata,
									   UnitOfMeasurement,
									   SUM (hi) hi,
									   SUM (shi) shi,
									   SUM (sbi) sbi
								  FROM (SELECT fcba,
											   FCCODE stockGenCode,
											   FCNAME stockGenName,
											   param_2 orderdata,
											   param_3 unitofmeasurement,
											   0 hi,
											   0 shi,
											   0 sbi
										  FROM PARAMETER
										 WHERE PARCODE = 'SR003_STOCKSALDO' AND PARAM_1 = 'DEFAULT'
										UNION
										  SELECT c.fcba,
												 c.stockGenCode,
												 c.stockGenName,
												 c.orderdata,
												 c.UnitOfMeasurement,
												 SUM (a.quantity) hi,
												 0 shi,
												 0 sbi
											FROM STOCKTRANSHIST a,
												 (SELECT a.fcba,
														 a.FCCODE stockGenCode,
														 a.FCNAME StockGenName,
														 a.param_2 orderdata,
														 a.param_3 unitofmeasurement,
														 b.param_1 stockcode
													FROM PARAMETER a, parameterdetail b
												   WHERE TRIM (a.fccode) = TRIM (b.parheadcode) AND a.parcode = b.parcode AND a.fcba = b.fcba AND a.PARCODE = 'SR003_STOCKSALDO' AND a.PARAM_1 = 'DEFAULT') c
										   WHERE a.stockcode = c.stockcode AND TO_NUMBER (TO_CHAR (a.transdate, 'YYYYMMDD')) <= TO_NUMBER (TO_CHAR (todate__, 'YYYYMMDD')) AND a.fcba = c.fcba
										GROUP BY c.fcba,
												 c.stockGenCode,
												 c.stockGenName,
												 c.orderdata,
												 c.UnitOfMeasurement
										UNION
										  SELECT c.fcba,
												 c.stockGenCode,
												 c.stockGenName,
												 c.orderdata,
												 c.UnitOfMeasurement,
												 0 hi,
												 SUM (a.quantity) shi,
												 0 sbi
											FROM STOCKTRANSHIST a,
												 (SELECT a.fcba,
														 a.FCCODE stockGenCode,
														 a.FCNAME StockGenName,
														 a.param_2 orderdata,
														 a.param_3 unitofmeasurement,
														 b.param_1 stockcode
													FROM PARAMETER a, parameterdetail b
												   WHERE TRIM (a.fccode) = TRIM (b.parheadcode) AND a.parcode = b.parcode AND a.fcba = b.fcba AND a.PARCODE = 'SR003_STOCKSALDO' AND a.PARAM_1 = 'DEFAULT') c
										   WHERE a.stockcode = c.stockcode AND TO_NUMBER (TO_CHAR (a.transdate, 'YYYYMMDD')) <= TO_NUMBER (TO_CHAR (todate__, 'YYYYMMDD')) AND a.fcba = c.fcba
										GROUP BY c.fcba,
												 c.stockGenCode,
												 c.stockGenName,
												 c.orderdata,
												 c.UnitOfMeasurement
										UNION
										  SELECT c.fcba,
												 c.stockGenCode,
												 c.stockGenName,
												 c.orderdata,
												 c.UnitOfMeasurement,
												 0 hi,
												 0 shi,
												 SUM (a.quantity) sbi
											FROM STOCKTRANSHIST a,
												 (SELECT a.fcba,
														 a.FCCODE stockGenCode,
														 a.FCNAME StockGenName,
														 a.param_2 orderdata,
														 a.param_3 unitofmeasurement,
														 b.param_1 stockcode
													FROM PARAMETER a, parameterdetail b
												   WHERE TRIM (a.fccode) = TRIM (b.parheadcode) AND a.parcode = b.parcode AND a.fcba = b.fcba AND a.PARCODE = 'SR003_STOCKSALDO' AND a.PARAM_1 = 'DEFAULT') c
										   WHERE a.stockcode = c.stockcode AND TO_NUMBER (TO_CHAR (a.transdate, 'YYYYMMDD')) <= TO_NUMBER (TO_CHAR (todate__, 'YYYYMMDD')) AND a.fcba = c.fcba
										GROUP BY c.fcba,
												 c.stockGenCode,
												 c.stockGenName,
												 c.orderdata,
												 c.UnitOfMeasurement)
							  GROUP BY fcba,
									   stockGenCode,
									   stockGenName,
									   orderdata,
									   UnitOfMeasurement)
					GROUP BY fcba,
							 nomer,
							 codedesc,
							 stockGenCode,
							 description,
							 orderdata,
							 UnitOfMeasurement)
			 WHERE fcba = fcba_;

		CURSOR getInfoSite
		IS
			SELECT b.groupplantation,
				   b.plantation,
				   b.groupregion,
				   b.region,
				   b.fccompanycode companycode
			  FROM businessunit b
			 WHERE B.FCCODE = fcba_;

		maxLineNo NUMBER;
		temp_date DATE;

		temp_groupplantationCode BUSINESSUNIT.GROUPPLANTATION%TYPE;
		temp_plantationCode BUSINESSUNIT.PLANTATION%TYPE;
		temp_groupregionCode BUSINESSUNIT.GROUPREGION%TYPE;
		temp_regionCode BUSINESSUNIT.REGION%TYPE;
		temp_CompanyCode BUSINESSUNIT.FCCOMPANYCODE%TYPE;
	BEGIN


		OPEN getMaxLineNo;

		FETCH getMaxLineNo INTO maxLineNo;

		CLOSE getMaxLineNo;

		IF maxLineNo = ''
		THEN
			maxLineNo := 0;
		END IF;


		OPEN getInfoSite;

		FETCH getInfoSite
		INTO temp_groupplantationCode, temp_plantationCode, temp_groupregionCode, temp_regionCode, temp_CompanyCode;

		CLOSE getInfoSite;


		temp_date := date_from_;



		LOOP
			EXIT WHEN TO_CHAR (temp_date, 'YYYYMMDD') > TO_CHAR (date_until_, 'YYYYMMDD');

			maxLineNo := NVL (maxLineNo, 0) + 1;



			-- Land Clearing
			FOR data__ IN setData (temp_date)
			LOOP
				INSERT INTO Production_Summary_Tab (LINENO,
													FCBA,
													STEPNUMBER,
													ONDATE,
													SEQNUMBER,
													CODEDESC,
													GROUPPLANTATIONCODE,
													PLANTATIONCODE,
													GROUPREGIONCODE,
													REGIONCODE,
													COMPANYCODE,
													JOBGENCODE,
													DESCRIPTION,
													UNITOFMEASUREMENT,
													TODATE,
													TOMONTH,
													TOYEAR,
													BUDGETTODATE,
													BUDGETTOMONTH,
													BUDGETTOYEAR)
					 VALUES (maxLineNo,
							 data__.FCBA,
							 nStep_,
							 temp_date,
							 data__.seqnumber,
							 data__.CODEDESC,
							 temp_groupplantationCode,
							 temp_plantationCode,
							 temp_groupregionCode,
							 temp_regionCode,
							 temp_CompanyCode,
							 data__.JOBGENCODE,
							 data__.DESCRIPTION,
							 data__.UNITOFMEASUREMENT,
							 data__.hi,
							 data__.shi,
							 data__.sbi,
							 data__.budget_hi,
							 data__.budget_shi,
							 data__.budget_sbi);


				maxLineNo := maxLineno + 1;


			END LOOP;

			temp_date := temp_date + 1;


			COMMIT;
		END LOOP;



	END EndingBalance_Process;



	PROCEDURE VehicleGroup_Process (fcba_			  IN VARCHAR2,
									date_from_		  IN DATE,
									date_until_ 	  IN DATE,
									budget_year_	  IN NUMBER,

							 codedesc,
							 orderreport,
							 description,
							 ROUND (SUM (NVL (hi, 0)), 2) hi,
							 ROUND (SUM (NVL (shi, 0)), 2) shi,
							 ROUND (SUM (NVL (sbi, 0)), 0) sbi,
							 ROUND (SUM (NVL (budget_hi, 0)), 2) budget_hi,
							 ROUND (SUM (NVL (budget_shi, 0)), 2) budget_shi,
							 ROUND (SUM (NVL (budget_sbi, 0)), 2) budget_sbi
						FROM (	SELECT fcba,
									   orderreport,
									   codedesc,
									   description,
									   SUM (hi) hi,
									   SUM (shi) shi,
									   SUM (sbi) sbi,
									   SUM (budget_hi) budget_hi,
									   SUM (budget_shi) budget_shi,
									   SUM (budget_sbi) budget_sbi
								  FROM (SELECT fcba,
											   orderreport,
											   'SR003_13' codedesc,
											   description,
											   0 hi,
											   0 shi,
											   0 sbi,
											   0 budget_hi,
											   0 budget_shi,
											   0 budget_sbi
										  FROM vehiclegroup
										UNION
										  SELECT fcba,
												 orderreport,
												 'SR003_13' codedesc,
												 description,
												 SUM (NVL (solar, 0)) hi,
												 0 shi,
												 0 sbi,
												 0 budget_hi,
												 0 budget_shi,
												 0 budget_sbi
											FROM (SELECT c.fcba,
														 c.orderreport,
														 c.vehiclegroupcode vehiclegroupcode,
														 c.description description,
														 a.vehiclecode vehiclecode,
														 b.fcname vehiclename,
														 a.unit unit,
														 a.hours hours,
														 a.solar solar,
														 a.solar rasio
													FROM view_vehicleproductivity a, VEHICLE b, VEHICLEGROUP c
												   WHERE	 TO_NUMBER (TO_CHAR (a.actiondate, 'YYYYMMDD')) = TO_NUMBER (TO_CHAR (todate__, 'YYYYMMDD'))
														 AND a.vehiclecode = b.fccode
														 AND b.vehiclegroupcode = c.vehiclegroupcode
														 AND a.fcba = b.fcba
														 AND b.fcba = c.fcba)
										GROUP BY fcba,
												 orderreport,
												 vehiclegroupcode,
												 description
										UNION
										  SELECT fcba,
												 orderreport,
												 'SR003_13' codedesc,
												 description,
												 0 hi,
												 SUM (NVL (solar, 0)) shi,
												 0 sbi,
												 0 budget_hi,
												 0 budget_shi,
												 0 budget_sbi
											FROM (SELECT c.fcba,
														 c.orderreport,
														 c.vehiclegroupcode vehiclegroupcode,
														 c.description description,
														 a.vehiclecode vehiclecode,
														 b.fcname vehiclename,
														 a.unit unit,
														 a.hours hours,
														 a.solar solar,
														 a.solar rasio
													FROM view_vehicleproductivity a, VEHICLE b, VEHICLEGROUP c
												   WHERE	 TO_NUMBER (TO_CHAR (a.actiondate, 'YYYYMMDD')) <= TO_NUMBER (TO_CHAR (todate__, 'YYYYMMDD'))
														 AND TO_NUMBER (TO_CHAR (a.actiondate, 'YYYY')) = TO_NUMBER (TO_CHAR (todate__, 'YYYY'))
														 AND TO_NUMBER (TO_CHAR (a.actiondate, 'MM')) = TO_NUMBER (TO_CHAR (todate__, 'MM'))
														 AND a.vehiclecode = b.fccode
														 AND b.vehiclegroupcode = c.vehiclegroupcode
														 AND a.fcba = b.fcba
														 AND b.fcba = c.fcba)
										GROUP BY fcba,
												 orderreport,
												 vehiclegroupcode,
												 description
										UNION
										  SELECT fcba,
												 orderreport,
												 'SR003_13' codedesc,
												 description,
												 0 hi,
												 0 shi,
												 SUM (NVL (solar, 0)) sbi,
												 0 budget_hi,
												 0 budget_shi,
												 0 budget_sbi
											FROM (SELECT c.fcba,
														 c.orderreport,
														 c.vehiclegroupcode vehiclegroupcode,
														 c.description description,
														 a.vehiclecode vehiclecode,
														 b.fcname vehiclename,
														 a.unit unit,
														 a.hours hours,
														 a.solar solar,
														 a.solar rasio
													FROM view_vehicleproductivity a, VEHICLE b, VEHICLEGROUP c
												   WHERE	 TO_NUMBER (TO_CHAR (a.actiondate, 'YYYYMMDD')) <= TO_NUMBER (TO_CHAR (todate__, 'YYYYMMDD'))
														 AND TO_NUMBER (TO_CHAR (a.actiondate, 'YYYY')) = TO_NUMBER (TO_CHAR (todate__, 'YYYY'))
														 AND a.vehiclecode = b.fccode
														 AND b.vehiclegroupcode = c.vehiclegroupcode
														 AND a.fcba = b.fcba
														 AND b.fcba = c.fcba)
										GROUP BY fcba,
												 orderreport,
												 vehiclegroupcode,
												 description)
							  GROUP BY fcba,
									   orderreport,
									   codedesc,
									   description)
					GROUP BY fcba,
							 orderreport,
							 codedesc,
							 description)
			 WHERE fcba = fcba_;

							 codedesc,

	PROCEDURE Fertilizer_Process (fcba_ 			IN VARCHAR2,

								  date_until_		IN DATE,

								  budget_version_	IN NUMBER,
								  module_version_	IN NUMBER,
								  nStep_			IN VARCHAR2)

		CURSOR getMaxLineNo
		IS

				   codedesc,
				   ROWNUM thenumber,
				   stockGenCode jobGenCode,
				   description,

				   orderdata seqnumber,
				   hi,
				   shi,
				   sbi,
				   budget_hi,
				   budget_shi,
				   budget_sbi
			  FROM (  SELECT fcba,
							 codedesc,
							 stockGenCode,
							 description,
							 orderdata,
							 UnitOfMeasurement,
							 SUM (hi) hi,
							 SUM (shi) shi,
							 ROUND (SUM (NVL (sbi, 0)), 2) sbi,
							 ROUND (SUM (NVL (budget_hi, 0)), 2) budget_hi,
							 ROUND (SUM (NVL (budget_shi, 0)), 2) budget_shi,
							 ROUND (SUM (NVL (budget_sbi, 0)), 2) budget_sbi
						FROM (	SELECT fcba,
									   '1' nomer,
									   'SR003_08' codedesc,
									   stockGenCode,
									   stockGenName description,
									   orderdata,

									   SUM (shi) shi,

											   FCCODE stockGenCode,

											   0 hi,

																								   '',
																								   '',
																								   '',
																								   '',
																								   todate__),
													0)
												   budget_sbi
										  FROM PARAMETER
										 WHERE PARCODE = 'SR003_PUPDET' AND PARAM_1 = 'DEFAULT'
										UNION
										  SELECT c.fcba,
												 c.stockGenCode,
												 c.stockGenName,
												 c.orderdata,

												 SUM (a.quantity) hi,
												 0 shi,
												 0 sbi,

	PROCEDURE HandlingAmount_Process (fcba_ 			IN VARCHAR2,
									  date_from_		IN DATE,
									  date_until_		IN DATE,
									  budget_year_		IN NUMBER,
									  budget_version_	IN NUMBER,
									  module_version_	IN NUMBER,
									  nStep_			IN VARCHAR2)
	IS
		CURSOR getMaxLineNo
		IS
			SELECT MAX (NVL (lineno, 0)) maxLineNo
			  FROM production_summary_tab
			 WHERE fcba = fcba_;

		CURSOR setData (todate__ IN DATE)
		IS
			SELECT codedesc,
				   fcba,
				   ROWNUM theNumber,

				   '' unitofmeasurement,
				   hi,
				   shi,
				   sbi,
				   budget_hi,
				   budget_shi,
				   budget_sbi
			  FROM (  SELECT codedesc,
							 fcba,
							 groupCode,
							 param_2_name,
							 ROUND (SUM (NVL (amount_hi_per_ha, 0)), 2) hi,
							 ROUND (SUM (NVL (amount_hi_per_ha_shi, 0)), 2) shi,
							 ROUND (SUM (NVL (amount_hi_per_ha_sbi, 0)), 2) sbi,
							 ROUND (SUM (NVL (budget_hi, 0)), 2) budget_hi,
							 ROUND (SUM (NVL (budget_shi, 0)), 2) budget_shi,
							 ROUND (SUM (NVL (budget_sbi, 0)), 2) budget_sbi
						FROM (	SELECT fcba,
									   'SR003_07_AMOUNT' codedesc,
									   '1' nomer,

	PROCEDURE Handling_Process (fcba_			  IN VARCHAR2,
								date_from_		  IN DATE,
								date_until_ 	  IN DATE,
								budget_year_	  IN NUMBER,
								budget_version_   IN NUMBER,
								module_version_   IN NUMBER,
								nStep_			  IN VARCHAR2)
	IS
		CURSOR getMaxLineNo
		IS
			SELECT MAX (NVL (lineno, 0)) maxLineNo
			  FROM production_summary_tab
			 WHERE fcba = fcba_;

		CURSOR setData (todate__ IN DATE)
		IS
			SELECT codedesc,
				   fcba,
				   ROWNUM theNumber,
				   groupCode JOBGENCODE,
				   description,
				   '' unitofmeasurement,
				   hi,
				   shi,
				   sbi,
				   budget_hi,
				   budget_shi,
				   budget_sbi
			  FROM (  SELECT codedesc,
							 fcba,
							 groupCode,
							 param_2_name description,
							 ROUND (SUM (NVL (hk_per_ha, 0)), 2) hi,
							 ROUND (SUM (NVL (hk_per_ha_shi, 0)), 2) shi,
							 ROUND (SUM (NVL (hk_per_ha_sbi, 0)), 2) sbi,
							 ROUND (SUM (NVL (budget_hk_per_ha, 0)), 2) budget_hi,
							 ROUND (SUM (NVL (budget_hk_per_ha_shi, 0)), 2) budget_shi,
							 ROUND (SUM (NVL (budget_hk_per_ha_sbi, 0)), 2) budget_sbi
						FROM (	SELECT fcba,
									   'SR003_06_MANDAYS' codedesc,
									   orderdata,
									   groupCode,
									   param_2_name,
									   SUM (total) hk_per_ha,
									   0 hk_per_ha_shi,
									   0 hk_per_ha_sbi,
									   0 budget_hk_per_ha,
									   0 budget_hk_per_ha_shi,
									   0 budget_hk_per_ha_sbi
								  FROM (SELECT b_temp.fcba,
											   b_temp.param_2,
											   b_temp.groupCode,
											   b_temp.orderdata,
											   b_temp.param_2_name,
											   0 sku,
											   0 khl,
											   0 total,
											   0 ha_harian,
											   0 bo_harian,

											   0 hk_per_ha,
											   0 gli_ltr,

											   0 par_ltr_ha,
											   0 met_kg,
											   0 met_kg_ha,

											   0 flo_kg_ha,

										  FROM JOB a_temp,
											   (SELECT grouphead.fcba,
													   grouphead.param_1,
													   grouphead.param_2,
													   groupname.fccode groupCode,
													   groupname.param_1 orderdata,
													   groupname.fcname param_2_name
												  FROM PARAMETERDETAIL grouphead,

															   fccode,
															   fcname,
															   param_1
														  FROM PARAMETERDETAIL
														 WHERE parcode = 'SR002' AND parheadcode = 'GROUPJOB') groupname
												 WHERE	   grouphead.parcode = 'SR002'
													   AND grouphead.parheadcode IN ('ACTIVITY_TBM', 'ACTIVITY_TM')
													   AND groupname.fcba = grouphead.fcba
													   AND groupname.fccode = grouphead.param_2) b_temp
										 WHERE a_temp.fccode = b_temp.param_1 AND a_temp.fcba = b_temp.fcba
										UNION
										  SELECT c.fcba,
												 c.param_2,
												 c.orderdata,
												 c.groupCode,
												 c.param_2_name,
												 SUM (a.sku) sku,
												 SUM (a.khl) khl,
												 SUM (a.total) total,
												 SUM (a.ha_harian) ha_harian,
												 SUM (a.bo_harian) bo_harian,
												 SUM (a.total_bo_harian) total_bo_harian,
												 DECODE (SUM (a.total), 0, 0, SUM (a.ha_harian) / SUM (a.total)) hk_per_ha,
												 SUM (a.gli_ltr) gli_ltr,
												 DECODE (SUM (a.total_bo_harian), 0, 0, SUM (a.gli_ltr) / SUM (a.total_bo_harian)) gli_ltr_ha,
												 SUM (a.par_ltr) par_ltr,

												 SUM (a.met_kg) met_kg,
												 DECODE (SUM (a.total_bo_harian), 0, 0, SUM (a.met_kg) / SUM (a.total_bo_harian)) met_kg_ha,
												 SUM (flo_kg) flo_kg,
												 DECODE (SUM (a.total_bo_harian), 0, 0, SUM (flo_kg) / SUM (a.total_bo_harian)) flo_kg_ha,
												 DECODE (SUM (a.ha_harian), 0, 0, SUM (a.amount_hi) / SUM (a.ha_harian)) amount_hi_per_ha

														 grouphead.param_1,
														 grouphead.param_2,

														 AND groupname.fcba = grouphead.fcba

										   WHERE	 (a.fcba = b.fcba AND a.jobcode = b.fccode)
												 AND (a.fcba = c.fcba AND a.jobcode = c.param_1)
												 AND TO_NUMBER (TO_CHAR (fddate, 'YYYYMMDD')) = TO_NUMBER (TO_CHAR (todate__, 'YYYYMMDD'))
										GROUP BY c.fcba,
												 c.param_2,
												 c.groupCode,
												 c.orderdata,
												 c.param_2_name
										ORDER BY param_2)
							  GROUP BY fcba,
									   param_2,
									   groupcode,
									   orderdata,
									   param_2_name
							  UNION
								SELECT fcba,
									   'SR003_06_MANDAYS' codedesc,
									   orderdata,
									   groupCode,
									   param_2_name,
									   0 hk_per_ha,
									   SUM (total) hk_per_ha_shi,
									   0 hk_per_ha_sbi,
									   0 budget_hk_per_ha,
									   0 budget_hk_per_ha_shi,
									   0 budget_hk_per_ha_sbi
								  FROM (SELECT b_temp.fcba,
											   b_temp.groupCode,
											   b_temp.param_2,
											   b_temp.orderdata,
											   b_temp.param_2_name,
											   0 sku,

											   0 total,

											   0 total_bo_harian,
											   0 hk_per_ha,
											   0 gli_ltr,

											   0 par_ltr_ha,

										  FROM JOB a_temp,

													   grouphead.param_1,
													   grouphead.param_2,
													   groupname.fccode groupCode,
													   groupname.param_1 orderdata,
													   groupname.fcname param_2_name
												  FROM PARAMETERDETAIL grouphead,
													   (SELECT fcba,
															   fccode,
															   fcname,
															   param_1
														  FROM PARAMETERDETAIL
														 WHERE parcode = 'SR002' AND parheadcode = 'GROUPJOB') groupname
												 WHERE	   grouphead.parcode = 'SR002'
													   AND grouphead.parheadcode IN ('ACTIVITY_TBM', 'ACTIVITY_TM')
													   AND groupname.fcba = grouphead.fcba
													   AND groupname.fccode = grouphead.param_2) b_temp
										 WHERE a_temp.fccode = b_temp.param_1 AND a_temp.fcba = b_temp.fcba
										UNION
										  SELECT c.fcba,
												 c.param_2,
												 c.orderdata,
												 c.groupCode,
												 c.param_2_name,
												 SUM (a.sku) sku,
												 SUM (a.khl) khl,
												 SUM (a.total) total,
												 SUM (a.ha_harian) ha_harian,
												 SUM (a.bo_harian) bo_harian,
												 SUM (a.total_bo_harian) total_bo_harian,
												 DECODE (SUM (a.total), 0, 0, SUM (a.ha_harian) / SUM (a.total)) hk_per_ha,
												 SUM (a.gli_ltr) gli_ltr,
												 DECODE (SUM (a.total_bo_harian), 0, 0, SUM (a.gli_ltr) / SUM (a.total_bo_harian)) gli_ltr_ha,

												 SUM (a.met_kg) met_kg,
												 DECODE (SUM (a.total_bo_harian), 0, 0, SUM (a.met_kg) / SUM (a.total_bo_harian)) met_kg_ha,
												 SUM (flo_kg) flo_kg,
												 DECODE (SUM (a.total_bo_harian), 0, 0, SUM (flo_kg) / SUM (a.total_bo_harian)) flo_kg_ha,
												 DECODE (SUM (a.ha_harian), 0, 0, SUM (a.amount_hi) / SUM (a.ha_harian)) amount_hi_per_ha
											FROM view_attendance_tm_smry a,
												 JOB b,
												 (SELECT grouphead.fcba,
														 grouphead.param_1,
														 grouphead.param_2,
														 groupname.fccode groupCode,
														 groupname.param_1 orderdata,

													FROM PARAMETERDETAIL grouphead,

														   WHERE parcode = 'SR002' AND parheadcode = 'GROUPJOB') groupname

														 AND groupname.fcba = grouphead.fcba
														 AND groupname.fccode = grouphead.param_2) c

	PROCEDURE Production_Process (fcba_ 			IN VARCHAR2,
								  date_from_		IN DATE,
								  date_until_		IN DATE,
								  budget_year_		IN NUMBER,
								  budget_version_	IN NUMBER,
								  module_version_	IN NUMBER,
								  nStep_			IN VARCHAR2)
	IS
		CURSOR getMaxLineNo
		IS
			SELECT MAX (NVL (lineno, 0)) maxLineNo
			  FROM production_summary_tab
			 WHERE fcba = fcba_;

		CURSOR setData (todate__ IN DATE)
		IS
			SELECT codedesc,
				   fcba,
				   ROWNUM theNumber,
				   '' JobGenCode,
				   description,
				   '' unitofmeasurement,
				   orderdata seqnumber,
				   hi,
				   shi,
				   sbi,
				   ROUND (budget_hi, 2) budget_hi,
				   ROUND (budget_shi, 2) budget_shi,
				   ROUND (budget_sbi, 2) budget_sbi
			  FROM (  SELECT codedesc,
							 fcba,
							 orderdata,
							 description,
							 ROUND (SUM (NVL (hi, 0)), 2) hi,
							 ROUND (SUM (NVL (shi, 0)), 2) shi,
							 ROUND (SUM (NVL (sbi, 0)), 2) sbi,
							 SUM (NVL (budget_hi, 0)) budget_hi,
							 SUM (NVL (budget_shi, 0)) budget_shi,
							 SUM (NVL (budget_sbi, 0)) budget_sbi
						FROM (SELECT fcba,
									 codedesc,
									 '1' thenumber,
									 (SELECT aa.fcname
										FROM PARAMETERDETAIL aa
									   WHERE aa.parcode = 'SR003' AND aa.ParheadCode = 'REPORTCODE' AND aa.fccode = codedesc AND aa.fcba = fcba AND ROWNUM = 1)
										 description,
									 (SELECT aa.param_3 orderdata
										FROM PARAMETERDETAIL aa
									   WHERE aa.parcode = 'SR003' AND aa.ParheadCode = 'REPORTCODE' AND aa.fccode = codedesc AND aa.fcba = fcba AND ROWNUM = 1)
										 orderdata,
									 hi,
									 shi,
									 sbi,
									 budget_hi,
									 budget_shi,
									 budget_sbi
								FROM (	SELECT fcba,
											   codedesc,
											   SUM (hi) hi,
											   SUM (shi) shi,
											   SUM (sbi) sbi,
											   SUM (budget_hi) budget_hi,
											   SUM (budget_shi) budget_shi,
											   SUM (budget_sbi) budget_sbi
										  FROM (  SELECT fcba,
														 codedesc,
														 SUM (hi) hi,
														 SUM (shi) shi,
														 SUM (sbi) sbi,
														 BUDGET_TRANSACTION_PRODQTY_API.GetQuantity_HI (fcba,
																										budget_year_,
																										budget_version_,
																										module_version_,
																										'',
																										'',
																										'',
																										todate__)
															 budget_hi,
														 BUDGET_TRANSACTION_PRODQTY_API.GetQuantity_SHI (fcba,
																										 budget_year_,
																										 budget_version_,
																										 module_version_,
																										 '',
																										 '',
																										 '',
																										 todate__)
															 budget_shi,
														 BUDGET_TRANSACTION_PRODQTY_API.GetQuantity_SBI (fcba,
																										 budget_year_,
																										 budget_version_,
																										 module_version_,
																										 '',
																										 '',
																										 '',
																										 todate__)
															 budget_sbi
													FROM (SELECT fccode fcba,
																 'SR003_01' AS codedesc,
																 0 hi,
																 0 shi,
																 0 sbi,
																 0 budget_hi,
																 0 budget_shi,
																 0 budget_sbi
															FROM businessunit
														  UNION
															SELECT fcba,
																   'SR003_01' AS codedesc,
																   SUM (totaldailyreceivetonase0) hi,
																   0 shi,
																   0 sbi,
																   0 budget_hi,
																   0 budget_shi,
																   0 budget_sbi
															  FROM view_dailyharvesting_mill
															 WHERE	   accountingyear = TO_CHAR (todate__, 'YYYY')
																   AND accountingperiod = TO_CHAR (todate__, 'MM')
																   AND TO_NUMBER (TO_CHAR (receptiondate, 'YYYYMMDD')) = TO_NUMBER (TO_CHAR (todate__, 'YYYYMMDD'))
														  GROUP BY fcba
														  UNION
															SELECT fcba,
																   'SR003_01' AS codedesc,
																   0 hi,
																   SUM (totaldailyreceivetonase0) shi,
																   0 sbi,
																   0 budget_hi,
																   0 budget_shi,
																   0 budget_sbi
															  FROM view_dailyharvesting_mill
															 WHERE	   accountingyear = TO_CHAR (todate__, 'YYYY')
																   AND accountingperiod = TO_CHAR (todate__, 'MM')
																   AND TO_NUMBER (TO_CHAR (receptiondate, 'YYYYMMDD')) <= TO_NUMBER (TO_CHAR (todate__, 'YYYYMMDD'))
														  GROUP BY fcba
														  UNION
															SELECT fcba,
																   'SR003_01' AS codedesc,
																   0 hi,
																   0 shi,
																   SUM (totaldailyreceivetonase0) sbi,
																   0 budget_hi,
																   0 budget_shi,
																   0 budget_sbi
															  FROM view_dailyharvesting_mill

														  GROUP BY fcba)
												GROUP BY fcba, codedesc
												UNION
												SELECT fccode fcba,
													   'SR003_02' AS codedesc,
													   0 hi,
													   0 shi,
													   0 sbi,
													   0 budget_hi,
													   0 budget_shi,
													   0 budget_sbi
												  FROM businessunit
												UNION
												  SELECT fcba,
														 'SR003_02' AS codedesc,
														 (SUM (variantemp) + SUM (restantemp)) hi,
														 (SUM (variantemp) + SUM (restantemp)) shi,
														 (SUM (variantemp) + SUM (restantemp)) sbi,
														 0 budget_hi,
														 0 budget_shi,
														 0 budget_sbi
													FROM (SELECT fcba, varian variantemp, 0 restantemp
															FROM view_dailyharvesting_varian
														   WHERE TO_NUMBER (TO_CHAR (dateprod, 'YYYYMMDD')) = TO_NUMBER (TO_CHAR (todate__, 'YYYYMMDD'))
														  UNION
															SELECT fcba, 0 variantemp, SUM (varian) restantemp
															  FROM view_dailyharvesting_varian
															 WHERE TO_NUMBER (TO_CHAR (dateprod, 'YYYYMMDD')) < TO_NUMBER (TO_CHAR (todate__, 'YYYYMMDD'))
														  GROUP BY fcba)

												UNION

														 SUM (shi) shi,
														 SUM (sbi) sbi,
														 SUM (budget_hi) budget_hi,
														 SUM (budget_shi) budget_shi,
														 SUM (budget_sbi) budget_sbi
													FROM (SELECT fccode fcba,
																 'SR003_03' AS codedesc,
																 0 hi,
																 0 shi,
																 0 sbi,
																 0 budget_hi,
																 0 budget_shi,

																   0 budget_sbi
															  FROM harvesting_interval_tab
															 WHERE TO_NUMBER (TO_CHAR (ondate, 'YYYYMMDD')) = TO_NUMBER (TO_CHAR (todate__, 'YYYYMMDD'))
														  GROUP BY fcba
														  UNION
															SELECT fcba,
																   'SR003_03' AS codedesc,
																   0 hi,
																   MAX (originalnumber) shi,
																   0 sbi,
																   0 budget_hi,
																   0 budget_shi,
																   0 budget_sbi
															  FROM harvesting_interval_tab
															 WHERE	   TO_NUMBER (TO_CHAR (ondate, 'YYYYMMDD')) <= TO_NUMBER (TO_CHAR (todate__, 'YYYYMMDD'))
																   AND TO_NUMBER (TO_CHAR (ondate, 'YYYY')) = TO_NUMBER (TO_CHAR (todate__, 'YYYY'))

														  GROUP BY fcba
														  UNION
															SELECT fcba,
																   'SR003_03' AS codedesc,
																   0 hi,
																   0 shi,
																   MAX (originalnumber) sbi,
																   0 budget_hi,
																   0 budget_shi,
																   0 budget_sbi
															  FROM harvesting_interval_tab

													FROM (SELECT fccode fcba,

															  FROM WEATHER

															SELECT fcba,
